    local function getDamage()   return wire.ports.Damage ~= 0 and wire.ports.Damage or DAMAGE end
    local function getRadius()   return wire.ports.Radius ~= 0 and wire.ports.Radius or RADIUS end
    local function getCooldown() return math.max(0, wire.ports.Cooldown ~= 0 and wire.ports.Cooldown or FIRE_DELAY) end

    --  :    
    local function fireParams()
        local speed = wire.ports.Speed ~= 0 and wire.ports.Speed or SPEED
        local c = chip()
        -- Both the muzzle position and shot direction follow the chip's current local +X axis.
        local pos = c:localToWorld(Vector(MUZZLE_OFF, 0, MUZZLE_UP))
        local dir = c:getForward()
        dir = dir / dir:getLength()
        return pos, dir, speed
    end

    local function shellAng(dir)
        local a = dir:getAngle()
        return Angle(a.p+MODEL_ANG.p, a.y+MODEL_ANG.y, a.r+MODEL_ANG.r)
    end

    local function predictTrajectory(pos, dir, speed)
        local points = { pos }
        local p, v = pos, dir * speed
        local dt = 0.05
        local sampleEvery = 6 -- 0.3 s between points keeps the HUD curve smooth without large net messages
        local steps = math.min(600, math.ceil(MAX_FLIGHT / dt))

        for i = 1, steps do
            v = v - Vector(0,0,GRAVITY*dt)
            local np = p + v*dt
            local tr = trace.line(p, np, { chip(), owner() })
            if tr.Hit then
                points[#points + 1] = tr.HitPos
                return points
            end
            p = np
            if i % sampleEvery == 0 then points[#points + 1] = p end
        end

        if steps % sampleEvery ~= 0 then points[#points + 1] = p end
        return points
    end

    local function applyBlastDamage(pos)
        local radius, damage = getRadius(), getDamage()
        local ok, list = pcall(find.inSphere, pos, radius)
        if not ok then return end
        for _, e in ipairs(list) do
            if e:isValid() then
                local cls = e:getClass()
                local isTarget = e:isPlayer() or e:isNPC()
                    or cls == "gmod_wire_explosive" or cls == "prop_physics"
                if isTarget then
                    local d = (e:getPos() - pos):getLength()
                    local dmg = damage * math.max(0, 1 - d / radius)
                    if dmg > 1 then
                        pcall(function() e:applyDamage(dmg, owner(), chip(), 64, pos) end)
                        if cls == "gmod_wire_explosive" then
                            pcall(wire.triggerInput, e, "Detonate", 1)
                        end
                    end
                end
            end
        end
    end

    local function explode(s, pos)
        shells[s.id] = nil
        s.holo:remove()
        net.start("arty_boom")
        net.writeUInt(s.id, 16)
        net.writeVector(pos)
        net.send()
        effect.create("Explosion", { origin = pos, scale = BOOM_SCALE })
        effect.create("HelicopterMegaBomb", { origin = pos })
        effect.beamRingPoint(pos, 0.5, 20, getRadius(), 24, 6, Color(255, 200, 120))
        pcall(applyBlastDamage, pos)
    end

    local function fire()
        local t = timer.curtime()
        if t - lastFire < getCooldown() then return end
        lastFire = t
        local pos, dir, speed = fireParams()
        local holo = hologram.create(pos, shellAng(dir), MODEL)
        if not holo then return end
        holo:setTrails(TRAIL_START, TRAIL_END, TRAIL_LIFE, TRAIL_MAT, TRAIL_COLOR, nil, true)
        shotId = (shotId + 1) % 65536
        shells[shotId] = { id = shotId, holo = holo, pos = pos, vel = dir * speed, born = t }
        net.start("arty_fire")
        net.writeUInt(shotId, 16)
        net.writeEntity(holo)
        net.writeVector(pos)
        net.writeVector(dir)
        net.send()
    end

    hook.add("think", "arty", function()
        local t = timer.curtime()
        local dt = math.min(t - lastThink, 0.1)
        lastThink = t

        -- Send the predicted arc only to the owner, and only while they are seated.
        local ow = owner()
        if ow and ow:isValid() and ow:inVehicle() and t - lastPredict > PREDICT_RATE then
            lastPredict = t
            local pos, dir, speed = fireParams()
            local points = predictTrajectory(pos, dir, speed)
            net.start("arty_predict")
            net.writeUInt(#points, 7)
            for _, point in ipairs(points) do
                net.writeVector(point)
            end
            net.send(ow)
        end

        if dt <= 0 then return end
        for _, s in pairs(shells) do
            s.vel = s.vel - Vector(0, 0, GRAVITY * dt)
            local np = s.pos + s.vel * dt
            local tr = trace.line(s.pos, np, { chip(), owner() })
            if tr.Hit or t - s.born > MAX_FLIGHT then
                explode(s, tr.Hit and tr.HitPos or np)
            else
                s.pos = np
                s.holo:setPos(np)
                s.holo:setAngles(shellAng(s.vel))
            end
        end
    end)

    hook.add("input", "arty", function(n, v) if n == "Fire" and v > 0 then fire() end end)
    hook.add("playersay", "arty", function(ply, text)
        if ply ~= owner() then return end
        if text ~= "!fire" and text:sub(1, 6) ~= "!fire " then return end
        fire()
        return ""
    end)
    print("M777: !fire /  Fire |    ")
end

    local fxSet
    local function pickFx(cands)
        if not fxSet or next(fxSet) == nil then
            fxSet = {}
            for _, n in ipairs(particleEffect.getList()) do fxSet[n] = true end
        end
        for _, n in ipairs(cands) do if fxSet[n] then return n end end
        return nil
    end
    local function pickMuzzleFireFx()
        local name = pickFx(MUZZLE_FIRE_FX)
        if name then return name end
        for _, candidate in ipairs(particleEffect.getList()) do
            local lower = string.lower(candidate)
            if string.find(lower, "muzzle", 1, true)
                and (string.find(lower, "flash", 1, true) or string.find(lower, "fire", 1, true)) then
                return candidate
            end
        end
        return nil
    end

    local function anchor(pos, ang)

        if h then h:setColor(Color(255,255,255,0)) end

    local function attach(h, name)
        if not h or not name or not hasPermission("particleEffect.attach") then return nil end
        local ok, fx = pcall(particleEffect.attach, h, name, 1)
        if ok then return fx end
        return nil

    local muzzleMaterials = {}
    local function loadMuzzleMaterial(path)
        if muzzleMaterials[path] then return muzzleMaterials[path] end
        local ok, mat = pcall(material.load, path)
        if ok and mat then muzzleMaterials[path] = mat end
        return ok and mat or nil
    end

    local function addMuzzleParticle(em, mat, pos, startSize, endSize, startAlpha, life, velocity, gravity, air, color)
        if not mat then return false end
        local ok, spawned = pcall(function()
            local p = em:add(mat, pos, startSize, endSize, 0, 0, startAlpha, 0, life)
            if not p then return false end
            p:setVelocity(velocity)
            p:setGravity(gravity)
            p:setAirResistance(air)
            p:setColor(color)
            return true
        end)
        return ok and spawned == true
    end

    local function muzzleBlast(pos, dir)
        local ang = dir:getAngle()
        local right = ang:getRight()
        local smokeMat = hasPermission("material.load", "particle/particle_smokegrenade")
            and loadMuzzleMaterial("particle/particle_smokegrenade") or nil
        local flameMat = hasPermission("material.load", "sprites/flamelet1")
            and loadMuzzleMaterial("sprites/flamelet1") or nil
        local em
        if (smokeMat or flameMat) and hasPermission("particle.create") then
            local emitterOK, created = pcall(particle.create, pos, true)
            if emitterOK then em = created end
        end

        local madeSmoke, madeFire = false, false
        if em and smokeMat then
            for i = 1, MUZZLE_SMOKE_COUNT do
                local life = math.random(24, 36) / 10
                local vel = right * math.random(-220, 220)
                          + dir * math.random(30, 220)
                          + Vector(0, 0, math.random(55, 180))
                if addMuzzleParticle(em, smokeMat, pos + dir * math.random(0, 18),
                    math.random(32, 56), math.random(190, 320), math.random(175, 225), life,
                    vel, Vector(0, 0, math.random(20, 55)), math.random(18, 32) / 10,
                    Color(115, 110, 105)) then
                    madeSmoke = true
                end
            end
        end

        if em and flameMat then
            for i = 1, MUZZLE_FIRE_COUNT do
                local life = math.random(12, 26) / 100
                local spread = right * math.random(-90, 90) + Vector(0, 0, math.random(-70, 100))
                local vel = dir * math.random(650, 1450) + spread
                if addMuzzleParticle(em, flameMat, pos + dir * math.random(0, 16),
                    math.random(28, 48), math.random(75, 145), 255, life,
                    vel, Vector(0, 0, math.random(20, 100)), 2.4,
                    Color(255, math.random(95, 190), 35)) then
                    madeFire = true
                end
            end
            for i = 1, 5 do
                if addMuzzleParticle(em, flameMat, pos + dir * 10, 90, 160, 255, 0.12,
                    dir * 260, Vector(0, 0, 35), 1.5, Color(255, 225, 145)) then
                    madeFire = true
                end
            end
        end
        if em then pcall(function() em:destroy() end) end

        -- Use installed particle systems if custom sprite materials are unavailable.
        local fireFx, smokeFx, h
        if not madeFire or not madeSmoke then
            h = anchor(pos, ang)
            if h then
                if not madeFire then fireFx = attach(h, pickMuzzleFireFx()) end
                if not madeSmoke then smokeFx = attach(h, pickFx(MUZZLE_SMOKE_FX)) end
                timer.simple(MUZZLE_FLASH_LIFE, function()
                    if fireFx and fireFx:isValid() then fireFx:destroy() end
                end)
                timer.simple(MUZZLE_SMOKE_LIFE, function()
                    if smokeFx and smokeFx:isValid() then smokeFx:destroy() end
                    if h and h:isValid() then h:remove() end
                end)
            end
        end
    end

    local pools = {}
    local function initPool(key, url, fmin, fmax, simple, vol)
        local p = { chans = {}, idx = 0 }
        pools[key] = p
        if url == "" then return end
        for i = 1, 2 do
            local ok, err = pcall(bass.loadURL, url, "3d noblock noplay", function(snd, e)
                if snd and snd:isValid() then
                    snd:setFade(fmin, fmax, simple)
                    snd:setVolume(vol)
                    p.chans[#p.chans + 1] = snd
                else
                    print("   (" .. key .. "): " .. tostring(e))
                end
            end)
            if not ok then print("bass (" .. key .. "): " .. tostring(err)) end
        end
    end
    initPool("shot_near", SOUND_SHOT,     400, NEAR_DIST, true, 1)
    initPool("shot_far",  SOUND_SHOT_FAR, 300, 6000, true, FAR_SHOT_VOL)
    initPool("boom",      SOUND_BOOM,     400, 200000, false, 1.5)

    local function engineFallback(kind, far, pos)
        if kind == "boom" then
            sound.emitSound("weapons/explode" .. math.random(3,5) .. ".wav", pos, 140, 80, 1)
        else
            sound.emitSound("weapons/rpg/rocketfire1.wav", pos, far and 140 or 90, far and 65 or 100, 1)
        end
    end

    local function playEvent(kind, pos)
        local lp = player()
        if not (lp and lp:isValid()) then lp = owner() end
        local lpPos = (lp and lp:isValid()) and lp:getPos() or pos
        local dist = (lpPos - pos):getLength()
        local key, far, delay, sndPos = "boom", false, 0, pos
        if kind ~= "boom" then
            far = dist > NEAR_DIST
            key = far and "shot_far" or "shot_near"
            delay = far and dist / SOUND_SPEED or 0
            if far then
                local d = pos - lpPos
                local len = d:getLength()
                if len > 0 then d = d / len end
                sndPos = lpPos + d * FAR_SHOT_OFFSET
            end
        end
        local function go()
            local p = pools[key]
            if p and #p.chans > 0 then
                p.idx = p.idx % #p.chans + 1
                local snd = p.chans[p.idx]
                if snd:isValid() then
                    snd:setPos(sndPos); snd:setTime(0); snd:play()
                    return
                end
            end
            engineFallback(kind, far, sndPos)
        end
        if delay > 0.02 then timer.simple(delay, go) else go() end
    end

            end
        end
    end)

    local function updateMarker()
        local seated = seatedOwner()
        if not seated then
            trajectory = nil
            predictPos = nil
            topDownFrameReady = false
        end

        if predictPos and seated then
            if not marker or not marker:isValid() then
                marker = hologram.create(predictPos, Angle(0,0,0), MARKER_MODEL)
                if marker then marker:setColor(MARKER_COLOR) end
            end
            if marker and marker:isValid() then marker:setPos(predictPos + Vector(0,0,2)) end
        else
            if marker and marker:isValid() then marker:remove() end
            marker = nil
        end
    end
    net.receive("arty_predict", function()
        local count = net.readUInt(7)
        local points = {}
        for i = 1, count do
            points[i] = net.readVector()
        end
        if count >= 2 and seatedOwner() then
            trajectory = points
            predictPos = points[count]
        else
            trajectory = nil
            predictPos = nil
        end
        updateMarker()
    end)
    hook.add("think", "arty_marker", updateMarker)

    hook.add("renderscene", "arty_topdown_scene", function()
        if not seatedOwner() or not trajectory or not predictPos or not topDownRTReady then return end
        if not hasTopDownCameraPermission() or render.isInRenderView() then return end
        if render.renderViewsLeft() <= 0 then return end

        local now = timer.realtime()
        if now - lastTopDownRender < TOPDOWN_UPDATE_RATE then return end
        lastTopDownRender = now

        local target = predictPos
        local selectedRT = false
        local ok = pcall(function()
            render.selectRenderTarget(TOPDOWN_RT)
            selectedRT = true
            render.clear(Color(12, 18, 26, 255), true, true)
            render.renderView({
                origin = target + Vector(0, 0, TOPDOWN_CAMERA_HEIGHT),
                angles = Angle(90, 0, 0),
                x = 0, y = 0, w = TOPDOWN_RT_SIZE, h = TOPDOWN_RT_SIZE,
                aspect = 1,
                ortho = {
                    left = -TOPDOWN_CAMERA_SPAN, right = TOPDOWN_CAMERA_SPAN,
                    top = TOPDOWN_CAMERA_SPAN, bottom = -TOPDOWN_CAMERA_SPAN
                },
                znear = 1, zfar = TOPDOWN_CAMERA_HEIGHT * 5,
                drawviewmodel = false, drawviewer = false,
                drawmonitors = false, dopostprocess = false
            })
        end)
        if selectedRT then pcall(function() render.selectRenderTarget() end) end
        if ok then topDownFrameReady = true end
    end)

    local function drawTopDownCamera()
        local screenW = render.getGameResolution()
        local boxSize, viewSize = 240, 200
        local boxX, boxY = screenW - boxSize - 18, 18
        local viewX = boxX + (boxSize - viewSize) * 0.5
        local viewY = boxY + 30
        local cameraPermission = hasTopDownCameraPermission()

        render.setColor(Color(7, 12, 19, 235))
        render.drawRect(boxX, boxY, boxSize, boxSize)
        render.setColor(Color(115, 145, 165, 240))
        render.drawRectOutline(boxX, boxY, boxSize, boxSize, 1)
        render.setColor(Color(238, 244, 250, 255))
        render.drawSimpleText(boxX + 10, boxY + 7, "LIVE IMPACT CAMERA", 0, 0)

        if cameraPermission and topDownFrameReady and trajectory and predictPos then
            render.setColor(Color(255, 255, 255, 255))
            render.setRenderTargetTexture(TOPDOWN_RT)
            render.drawTexturedRectUV(viewX, viewY, viewSize, viewSize, 0, 0, 0.25, 0.25)
            render.setRenderTargetTexture(nil)

            local centerX, centerY = viewX + viewSize * 0.5, viewY + viewSize * 0.5
            render.setColor(Color(255, 58, 48, 255))
            render.drawLine(centerX - 10, centerY, centerX + 10, centerY)
            render.drawLine(centerX, centerY - 10, centerX, centerY + 10)
            render.drawCircle(centerX, centerY, 6)
        else
            render.setColor(Color(12, 18, 26, 255))
            render.drawRect(viewX, viewY, viewSize, viewSize)
            render.setColor(Color(225, 232, 240, 255))
            local status = "CAMERA LOADING"
            if not topDownRTReady then
                status = "CAMERA UNAVAILABLE"
            elseif not cameraPermission then
                status = "ALLOW CAMERA"
            elseif not trajectory or not predictPos then
                status = "WAITING FOR AIM"
            end
            render.drawSimpleText(viewX + viewSize * 0.5, viewY + viewSize * 0.5, status, 1, 1)
        end

        render.setColor(Color(115, 145, 165, 240))
        render.drawRectOutline(viewX, viewY, viewSize, viewSize, 1)
        if trajectory and #trajectory >= 2 then
            local first, impact = trajectory[1], trajectory[#trajectory]
            local dx, dy = impact.x - first.x, impact.y - first.y
            local rangeMeters = math.floor(math.sqrt(dx * dx + dy * dy) / 52.5)
            render.setColor(Color(238, 244, 250, 255))
            render.drawSimpleText(boxX + 10, boxY + boxSize - 17, "RANGE " .. rangeMeters .. " m", 0, 0)
        end
    end

    end)

    local trails, dead = {}, {}
    net.receive("arty_fire", function()
        local id = net.readUInt(16)
        net.readEntity(function(holo)
            if not holo or dead[id] then return end
            local ok, fx = pcall(particleEffect.attach, holo, SMOKE_FX, 1)
            if ok then trails[id] = fx end
            local ok2, ws = pcall(sound.create, holo, WHISTLE_PATH)
            if ok2 and ws then
                pcall(function()
                    ws:setSoundLevel(WHISTLE_LEVEL)
                    ws:setVolume(WHISTLE_VOL)
                    ws:play()
                end)
                whistleActive[id] = { snd = ws, holo = holo, prev = holo:getPos() }
            end
        end)
        local pos = net.readVector()
        local dir = net.readVector()
        local ok, err = pcall(muzzleBlast, pos, dir)
        if not ok then print("muzzleBlast: " .. tostring(err)) end
        pcall(playEvent, "shot", pos)
    end)

    net.receive("arty_boom", function()
        local id = net.readUInt(16)
        local pos = net.readVector()
        dead[id] = true
        timer.simple(10, function() dead[id] = nil end)
        local fx = trails[id]; trails[id] = nil
        if fx and fx:isValid() then
            fx:stopEmission()
            timer.simple(4, function() if fx:isValid() then fx:destroy() end end)
        end
        local w = whistleActive[id]; whistleActive[id] = nil
        if w then pcall(function() w.snd:stop(); w.snd:destroy() end) end
        pcall(bigBoomFx, pos)
        pcall(playEvent, "boom", pos)
    end)
end
