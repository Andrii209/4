    -- Connect Wire inputs Damage and Radius to customize the blast; 0 keeps the defaults above.
    local function getDamage()
        local value = wire.ports.Damage
        if value == 0 then return DAMAGE end
        return math.max(0, math.min(value, DAMAGE_LIMIT))
    end
    local function getRadius()
        local value = wire.ports.Radius
        if value == 0 then return RADIUS end
        return math.max(1, math.min(value, RADIUS_LIMIT))
    end
    local function getCooldown() return math.max(0, wire.ports.Cooldown ~= 0 and wire.ports.Cooldown or FIRE_DELAY) end

    --  :    
    local function fireParams()
        local speed = wire.ports.Speed ~= 0 and wire.ports.Speed or SPEED
        local c = chip()
        -- Both the muzzle position and shot direction follow the chip's current local +X axis.
        local pos = c:localToWorld(Vector(MUZZLE_OFF, 0, MUZZLE_UP))
        local dir = c:getForward()
        dir = dir / dir:getLength()
        return pos, dir, speed
    end

    local function shellAng(dir)
        local a = dir:getAngle()
        return Angle(a.p+MODEL_ANG.p, a.y+MODEL_ANG.y, a.r+MODEL_ANG.r)
    end

    local function predictTrajectory(pos, dir, speed)
        local points = { pos }
        local p, v = pos, dir * speed
        local dt = 0.075
        local sampleEvery = 4 -- Keep 0.3 s HUD spacing while using fewer collision traces.
        local steps = math.min(270, math.ceil(MAX_FLIGHT / dt))

        for i = 1, steps do
            v = v - Vector(0,0,GRAVITY*dt)
            local np = p + v*dt
            local tr = trace.line(p, np, { chip(), owner() })
            if tr.Hit then
                points[#points + 1] = tr.HitPos
                return points
            end
            p = np
            if i % sampleEvery == 0 then points[#points + 1] = p end
        end

        if steps % sampleEvery ~= 0 then points[#points + 1] = p end
        return points
    end

    local function applyBlastDamage(pos)
        local radius, damage = getRadius(), getDamage()
        if damage <= 0 then return end
        local ok, list = pcall(find.inSphere, pos, radius)
        if not ok then return end
        for _, e in ipairs(list) do
            if e:isValid() then
                local cls = e:getClass()
                local isTarget = e:isPlayer() or e:isNPC()
                    or cls == "gmod_wire_explosive" or cls == "prop_physics"
                if isTarget then
                    local d = (e:getPos() - pos):getLength()
                    local dmg = damage * math.max(0, 1 - d / radius)
                    if dmg > 0 then
                        pcall(function() e:applyDamage(dmg, owner(), chip(), 64, pos) end)
                        if cls == "gmod_wire_explosive" then
                            pcall(wire.triggerInput, e, "Detonate", 1)
                        end
                    end
                end
            end
        end
    end

    local function explode(s, pos)
        shells[s.id] = nil
        s.holo:remove()
        net.start("arty_boom")
        net.writeUInt(s.id, 16)
        net.writeVector(pos)
        net.send()
        effect.create("Explosion", { origin = pos, scale = BOOM_SCALE, magnitude = BOOM_SCALE })
        effect.create("HelicopterMegaBomb", { origin = pos, scale = BOOM_SCALE, magnitude = BOOM_SCALE })
        local shockwaveRadius = math.max(getRadius(), 1400)
        effect.beamRingPoint(pos, 1.1, 45, shockwaveRadius, 52, 12, Color(255, 190, 115))
        pcall(applyBlastDamage, pos)
    end

    local function fire()
        local t = timer.curtime()
        if t - lastFire < getCooldown() then return end
        lastFire = t
        local pos, dir, speed = fireParams()
        local holo = hologram.create(pos, shellAng(dir), MODEL)
        if not holo then return end
        holo:setTrails(TRAIL_START, TRAIL_END, TRAIL_LIFE, TRAIL_MAT, TRAIL_COLOR, nil, true)
        shotId = (shotId + 1) % 65536
        shells[shotId] = { id = shotId, holo = holo, pos = pos, vel = dir * speed, born = t }
        net.start("arty_fire")
        net.writeUInt(shotId, 16)
        net.writeEntity(holo)
        net.writeVector(pos)
        net.writeVector(dir)
        net.send()
    end

    hook.add("think", "arty", function()
        local t = timer.curtime()
        local dt = math.min(t - lastThink, 0.1)
        lastThink = t

        -- The server owns trajectory prediction; send it to the chip owner for their client camera.
        local ow = owner()
        if ow and ow:isValid() and t - lastPredict > PREDICT_RATE then
            lastPredict = t
            local pos, dir, speed = fireParams()
            local points = predictTrajectory(pos, dir, speed)
            net.start("arty_predict")
            net.writeUInt(#points, 7)
            for _, point in ipairs(points) do
                net.writeVector(point)
            end
            net.send(ow)
        end

        if dt <= 0 then return end
        for _, s in pairs(shells) do
            s.vel = s.vel - Vector(0, 0, GRAVITY * dt)
            local np = s.pos + s.vel * dt
            local tr = trace.line(s.pos, np, { chip(), owner() })
            if tr.Hit or t - s.born > MAX_FLIGHT then
                explode(s, tr.Hit and tr.HitPos or np)
            else
                s.pos = np
                s.holo:setPos(np)
                s.holo:setAngles(shellAng(s.vel))
            end
        end
    end)

    hook.add("input", "arty", function(n, v) if n == "Fire" and v > 0 then fire() end end)
    hook.add("playersay", "arty", function(ply, text)
        if ply ~= owner() then return end
        if text ~= "!fire" and text:sub(1, 6) ~= "!fire " then return end
        fire()
        return ""
    end)
    print("M777: Fire / !fire | Wire Damage and Radius inputs (0 = defaults)")
end

    local function getFxSet()
        if not fxSet or next(fxSet) == nil then
            fxSet = {}
            for _, n in ipairs(particleEffect.getList()) do fxSet[string.lower(n)] = n end
        end
        return fxSet
    end
    local function pickFx(cands)
        local available = getFxSet()
        for _, n in ipairs(cands) do
            local found = available[string.lower(n)]
            if found then return found end
        end
        return nil
    end
    local function pickFxContaining(word, excluded)
        local available = getFxSet()
        for lower, original in pairs(available) do
            if string.find(lower, word, 1, true)
                and (not excluded or not string.find(lower, excluded, 1, true)) then
                return original
            end
        end
        return nil
    end

    local pools = {}
    local function initPool(key, url, fmin, fmax, simple, vol)
        local p = { chans = {}, idx = 0 }
        pools[key] = p
        if url == "" then return end
        for i = 1, 2 do
            local ok, err = pcall(bass.loadURL, url, "3d noblock noplay", function(snd, e)
                if snd and snd:isValid() then
                    snd:setFade(fmin, fmax, simple)
                    snd:setVolume(vol)
                    p.chans[#p.chans + 1] = snd
                else
                    print("   (" .. key .. "): " .. tostring(e))
                end
            end)
            if not ok then print("bass (" .. key .. "): " .. tostring(err)) end
        end
    end
    initPool("shot_near", SOUND_SHOT,     400, NEAR_DIST, true, 1)
    initPool("shot_far",  SOUND_SHOT_FAR, 300, 6000, true, FAR_SHOT_VOL)
    initPool("boom",      SOUND_BOOM,     400, 200000, false, 1.5)

    local function engineFallback(kind, far, pos)
        if kind == "boom" then
            sound.emitSound("weapons/explode" .. math.random(3,5) .. ".wav", pos, 140, 80, 1)
        else
            sound.emitSound("weapons/rpg/rocketfire1.wav", pos, far and 140 or 90, far and 65 or 100, 1)
        end
    end

    local function playEvent(kind, pos)
        local lp = player()
        if not (lp and lp:isValid()) then lp = owner() end
        local lpPos = (lp and lp:isValid()) and lp:getPos() or pos
        local dist = (lpPos - pos):getLength()
        local key, far, delay, sndPos = "boom", false, 0, pos
        if kind ~= "boom" then
            far = dist > NEAR_DIST
            key = far and "shot_far" or "shot_near"
            delay = far and dist / SOUND_SPEED or 0
            if far then
                local d = pos - lpPos
                local len = d:getLength()
                if len > 0 then d = d / len end
                sndPos = lpPos + d * FAR_SHOT_OFFSET
            end
        end
        local function go()
            local p = pools[key]
            if p and #p.chans > 0 then
                p.idx = p.idx % #p.chans + 1
                local snd = p.chans[p.idx]
                if snd:isValid() then
                    snd:setPos(sndPos); snd:setTime(0); snd:play()
                    return
                end
            end
            engineFallback(kind, far, sndPos)
        end
        if delay > 0.02 then timer.simple(delay, go) else go() end
    end

    -- ===== Impact camera: hold G to replace the normal view with a top-down view. =====
    local trajectory, predictPos, marker
    local function cameraOwner()
        local lp = player()
        return lp and lp:isValid() and lp == owner()
    end

    local permissionSetup, cameraPromptSent, lastPermissionTry = false, false, 0
    local cameraHudMissing = false
    local function isOwnerClient()
        local lp = player()
        return lp and lp:isValid() and lp == owner()
    end
    local function setupImpactCameraRequest()
        if permissionSetup or not isOwnerClient() then return end
        local ok = pcall(setupPermissionRequest,
            { "render.calcview", "input", "particle.create", "material.load", "particleEffect.attach" },
            "Hold G for the impact view and allow the muzzle and impact particles.", true
        )
        if ok then permissionSetup = true end
    end
    setupImpactCameraRequest()

    local function hasImpactCameraPermission()
        return permissionSetup and isOwnerClient()
            and hasPermission("render.calcview") and hasPermission("input")
    end
    local function hasImpactFxPermissions()
        return hasPermission("particle.create")
            and hasPermission("material.load", IMPACT_SMOKE_MAT)
            and hasPermission("material.load", IMPACT_FLAME_MAT)
    end

    local function impactCameraHeld()
        if not hasImpactCameraPermission() or not cameraOwner() or not predictPos then return false end
        local ok, down = pcall(input.isKeyDown, KEY.G)
        return ok and down == true
    end

    hook.add("think", "arty_impact_camera_permission", function()
        if not isOwnerClient() or not cameraOwner() then return end
        setupImpactCameraRequest()
        if cameraPromptSent or (hasImpactCameraPermission() and hasImpactFxPermissions()) then return end
        local now = timer.realtime()
        if now - lastPermissionTry < 1 then return end
        lastPermissionTry = now
        local ok, err = pcall(sendPermissionRequest)
        if ok then
            cameraPromptSent = true
            cameraHudMissing = false
        else
            cameraHudMissing = string.find(string.lower(tostring(err)), "hud", 1, true) ~= nil
        end
    end)

    hook.add("inputbindpressed", "arty_hold_g_context_menu", function(bind)
        if bind ~= "+menu" or not isOwnerClient() or not cameraOwner() or not predictPos then return end
        local ok, down = pcall(input.isKeyDown, KEY.G)
        if ok and down then return true end
    end)

    local impactCameraOffset = Vector(0, 0, IMPACT_CAMERA_HEIGHT)
    local impactCameraAngles = Angle(90, 0, 0)
    local impactCameraPosition, impactCameraLastTime
    hook.add("calcview", "arty_impact_camera", function()
        if not impactCameraHeld() then
            impactCameraPosition = nil
            impactCameraLastTime = nil
            return
        end

        local now = timer.realtime()
        local target = predictPos + impactCameraOffset
        if not impactCameraPosition then
            impactCameraPosition = target
        else
            local dt = math.min(math.max(now - (impactCameraLastTime or now), 0), 0.1)
            local blend = 1 - math.exp(-IMPACT_CAMERA_SMOOTHING * dt)
            impactCameraPosition = impactCameraPosition + (target - impactCameraPosition) * blend
        end
        impactCameraLastTime = now

        -- Use a tight perspective view and clip distant geometry to keep the live scene light.
        return {
            origin = impactCameraPosition,
            angles = impactCameraAngles,
            fov = IMPACT_CAMERA_FOV,
            znear = IMPACT_CAMERA_ZNEAR,
            zfar = IMPACT_CAMERA_ZFAR,
            drawviewer = false
        }
    end)

    local function updateMarker(cameraActive)
        local ownerActive = cameraOwner()
        if not ownerActive then
            trajectory = nil
            predictPos = nil
        end

        if predictPos and ownerActive then
            local markerPos = predictPos
            if cameraActive and impactCameraPosition then
                markerPos = impactCameraPosition - impactCameraOffset
            end
            if not marker or not marker:isValid() then
                marker = hologram.create(markerPos, Angle(0,0,0), MARKER_MODEL)
                if marker then marker:setColor(MARKER_COLOR) end
            end
            if marker and marker:isValid() then marker:setPos(markerPos + Vector(0,0,2)) end
        else
            if marker and marker:isValid() then marker:remove() end
            marker = nil
        end
    end
    net.receive("arty_predict", function()
        local count = net.readUInt(7)
        local points = {}
        for i = 1, count do
            points[i] = net.readVector()
        end
        if count >= 2 and cameraOwner() then
            trajectory = points
            predictPos = points[count]
        else
            trajectory = nil
            predictPos = nil
        end
        updateMarker(impactCameraHeld())
    end)
    local nextMarkerUpdate = 0
    hook.add("think", "arty_marker", function()
        local now = timer.realtime()
        if now < nextMarkerUpdate then return end
        local cameraActive = impactCameraHeld()
        nextMarkerUpdate = now + (cameraActive and 0.05 or PREDICT_RATE)
        updateMarker(cameraActive)
    end)

            end
            render.setColor(Color(238, 244, 250, 255))
            render.drawSimpleText(screenW - 18, 18, status, 2, 0)

            if not trajectory or #trajectory < 2 or not predictPos then return end
            local first, impact = trajectory[1], trajectory[#trajectory]
            local dx, dy = impact.x - first.x, impact.y - first.y
            local rangeMeters = math.floor(math.sqrt(dx * dx + dy * dy) / 52.5)
            render.drawSimpleText(screenW - 18, 38, "RANGE " .. rangeMeters .. " m", 2, 0)

            if cameraActive then
                local centerX, centerY = screenW * 0.5, screenH * 0.5
                render.setColor(Color(255, 58, 48, 255))
                render.drawLine(centerX - 12, centerY, centerX + 12, centerY)
                render.drawLine(centerX, centerY - 12, centerX, centerY + 12)
                render.drawCircle(centerX, centerY, 7)
                return
            end

            render.setColor(Color(255,190,70,220))
            for i = 1, #trajectory - 1 do
                local a = trajectory[i]:toScreen()
                local b = trajectory[i + 1]:toScreen()
                if a and b and a.visible and b.visible then
                    render.drawLine(a.x, a.y, b.x, b.y)
                end
            end

            local scr = predictPos:toScreen()
            if scr and scr.visible then
                render.setColor(Color(0,255,0,220))
                render.drawRect(scr.x-14, scr.y-2, 28, 4)
                render.drawRect(scr.x-2, scr.y-14, 4, 28)
                local m = math.floor((player():getPos()-predictPos):getLength() / 52.5)
                render.drawSimpleText(scr.x+16, scr.y-8, m .. " m", 0, 0)
            end
        end)
    end)

    local trails, dead = {}, {}
    net.receive("arty_fire", function()
        local id = net.readUInt(16)
        net.readEntity(function(holo)
            if not holo or dead[id] then return end
            local ok, fx = pcall(particleEffect.attach, holo, SMOKE_FX, 1)
            if ok then trails[id] = fx end
            local ok2, ws = pcall(sound.create, holo, WHISTLE_PATH)
            if ok2 and ws then
                pcall(function()
                    ws:setSoundLevel(WHISTLE_LEVEL)
                    ws:setVolume(WHISTLE_VOL)
                    ws:play()
                end)
                whistleActive[id] = { snd = ws, holo = holo, prev = holo:getPos() }
            end
        end)
        local pos = net.readVector()
        local dir = net.readVector()
        pcall(muzzleFlashFx, pos, dir)
        pcall(playEvent, "shot", pos)
    end)

    net.receive("arty_boom", function()
        local id = net.readUInt(16)
        local pos = net.readVector()
        dead[id] = true
        timer.simple(10, function() dead[id] = nil end)
        local fx = trails[id]; trails[id] = nil
        if fx and fx:isValid() then
            fx:stopEmission()
            timer.simple(4, function() if fx:isValid() then fx:destroy() end end)
        end
        local w = whistleActive[id]; whistleActive[id] = nil
        if w then pcall(function() w.snd:stop(); w.snd:destroy() end) end
        pcall(bigBoomFx, pos)
        pcall(playEvent, "boom", pos)
    end)
end
